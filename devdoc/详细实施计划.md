### 1.2 核心模块职责

| 模块 | 职责 | 主要功能 |
|------|------|----------|
| `users` | 用户和权限管理 | 用户认证、角色管理、权限控制 |
| `events` | 活动项目管理 | 活动CRUD、预算管理、看板数据 |
| `tasks` | 任务管理 | 任务CRUD、依赖关系、状态管理 |
| `checklists` | 核验清单管理 | 模板、实例、核验操作 |
| `profiles` | 关联方档案 | 档案CRUD、标签管理、历史记录 |
| `knowledge` | 知识库 | 知识条目、推荐算法、相似度计算 |
| `reviews` | 复盘管理 | 复盘模板、报告生成、数据分析 |

```python
# apps/users/models/user.py
from django.contrib.auth.models import AbstractUser
from django.db import models
import uuid

class User(AbstractUser):
    """扩展的用户模型"""
    id = models.UUIDField(primary_key=True, default=uuid.uuid4, editable=False)
    phone = models.CharField(max_length=20, blank=True)
    avatar_url = models.URLField(blank=True)
    department = models.CharField(max_length=100, blank=True)
    position = models.CharField(max_length=100, blank=True)
    is_active = models.BooleanField(default=True)
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
    
    class Meta:
        db_table = 'users'
        indexes = [
            models.Index(fields=['email']),
            models.Index(fields=['phone']),
        ]

class UserRole(models.Model):
    """用户角色"""
    ROLE_CHOICES = [
        ('admin', '系统管理员'),
        ('project_owner', '项目负责人'),
        ('executor', '执行成员'),
        ('observer', '只读观察者'),
    ]
    
    user = models.OneToOneField(User, on_delete=models.CASCADE, related_name='role')
    role = models.CharField(max_length=20, choices=ROLE_CHOICES, default='executor')
    created_at = models.DateTimeField(auto_now_add=True)
    
    class Meta:
        db_table = 'user_roles'

class ResourceAccess(models.Model):
    """资源级权限"""
    user = models.ForeignKey(User, on_delete=models.CASCADE, related_name='resource_accesses')
    resource_type = models.CharField(max_length=50)  # 'event', 'task', etc.
    resource_id = models.UUIDField()
    permission_level = models.PositiveSmallIntegerField(default=1)  # 1-4
    granted_by = models.ForeignKey(User, on_delete=models.SET_NULL, null=True, related_name='granted_permissions')
    created_at = models.DateTimeField(auto_now_add=True)
    expires_at = models.DateTimeField(null=True, blank=True)
    
    class Meta:
        db_table = 'resource_access'
        unique_together = ['user', 'resource_type', 'resource_id']
        indexes = [
            models.Index(fields=['resource_type', 'resource_id']),
            models.Index(fields=['user', 'created_at']),
        ]
```

```python
from django.db import models
import uuid

class Event(models.Model):
    """活动项目"""
    class Status(models.TextChoices):
        PLANNING = 'planning', '策划中'
        EXECUTING = 'executing', '执行中'
        COMPLETED = 'completed', '已完成'
        REVIEWED = 'reviewed', '已复盘'
        CANCELLED = 'cancelled', '已取消'
    
    id = models.UUIDField(primary_key=True, default=uuid.uuid4, editable=False)
    name = models.CharField(max_length=255)
    type = models.CharField(max_length=50)  # 培训、沙龙、年会等
    description = models.TextField(blank=True)
    
    # 时间信息
    start_date = models.DateTimeField()
    end_date = models.DateTimeField()
    
    # 客户信息
    client = models.CharField(max_length=255, blank=True)
    client_contact = models.CharField(max_length=100, blank=True)
    
    # 预算信息
    estimated_budget = models.DecimalField(max_digits=10, decimal_places=2, default=0)
    actual_budget = models.DecimalField(max_digits=10, decimal_places=2, default=0)
    budget_variance = models.DecimalField(max_digits=10, decimal_places=2, default=0)
    
    # 状态和归属
    status = models.CharField(max_length=20, choices=Status.choices, default=Status.PLANNING)
    owner = models.ForeignKey('users.User', on_delete=models.PROTECT, related_name='owned_events')
    
    # 元数据
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
    completed_at = models.DateTimeField(null=True, blank=True)
    
    class Meta:
        db_table = 'events'
        indexes = [
            models.Index(fields=['status']),
            models.Index(fields=['start_date']),
            models.Index(fields=['owner']),
            models.Index(fields=['-created_at']),
        ]
    
    def __str__(self):
        return f"{self.name} ({self.type})"
```

```python
# apps/tasks/models/task.py
from django.db import models
import uuid

class Task(models.Model):
    """任务"""
    class Status(models.TextChoices):
        PENDING = 'pending', '待办'
        READY = 'ready', '就绪'
        IN_PROGRESS = 'in_progress', '进行中'
        COMPLETED = 'completed', '已完成'
        CANCELLED = 'cancelled', '已取消'
        BLOCKED = 'blocked', '阻塞'
    
    class TaskType(models.TextChoices):
        PLANNING = 'planning', '策划'
        GUEST = 'guest', '嘉宾'
        MATERIAL = 'material', '物料'
        VENUE = 'venue', '场地'
        PROMOTION = 'promotion', '宣传'
        ONSITE = 'onsite', '现场'
        REVIEW = 'review', '复盘'
    
    id = models.UUIDField(primary_key=True, default=uuid.uuid4, editable=False)
    event = models.ForeignKey('events.Event', on_delete=models.CASCADE, related_name='tasks')
    
    # 基本信息
    title = models.CharField(max_length=255)
    description = models.TextField(blank=True)
    task_type = models.CharField(max_length=20, choices=TaskType.choices)
    
    # 责任和状态
    assignee = models.ForeignKey('users.User', on_delete=models.SET_NULL, null=True, blank=True, related_name='assigned_tasks')
    status = models.CharField(max_length=20, choices=Status.choices, default=Status.PENDING)
    progress = models.PositiveSmallIntegerField(default=0)  # 0-100
    
    # 时间信息
    due_date = models.DateTimeField(null=True, blank=True)
    start_date = models.DateTimeField(null=True, blank=True)
    completed_at = models.DateTimeField(null=True, blank=True)
    
    # 依赖关系（通过TaskDependency模型处理）
    
    # 元数据
    created_by = models.ForeignKey('users.User', on_delete=models.SET_NULL, null=True, blank=True)
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
    
    class Meta:
        db_table = 'tasks'
        indexes = [
            models.Index(fields=['event']),
            models.Index(fields=['status']),
            models.Index(fields=['task_type']),
            models.Index(fields=['assignee']),
            models.Index(fields=['due_date']),
            models.Index(fields=['-created_at']),
        ]

class TaskDependency(models.Model):
    """任务依赖关系"""
    task = models.ForeignKey(Task, on_delete=models.CASCADE, related_name='dependencies')
    depends_on = models.ForeignKey(Task, on_delete=models.CASCADE, related_name='dependents')
    created_at = models.DateTimeField(auto_now_add=True)
    
    class Meta:
        db_table = 'task_dependencies'
        unique_together = ['task', 'depends_on']
        indexes = [
            models.Index(fields=['task']),
            models.Index(fields=['depends_on']),
        ]

class CommunicationTask(models.Model):
    """沟通任务"""
    id = models.UUIDField(primary_key=True, default=uuid.uuid4, editable=False)
    task = models.OneToOneField(Task, on_delete=models.CASCADE, related_name='communication')
    
    # 沟通内容
    content = models.TextField()
    requirements = models.TextField(blank=True)
    
    # 沟通对象
    communicators = models.JSONField(default=list)  # 联系人列表
    
    # 结论性文件
    conclusion_files = models.JSONField(default=list)  # 文件引用列表
    is_closed = models.BooleanField(default=False)
    
    class Meta:
        db_table = 'communication_tasks'
```

```python
def check_permission(resource_type, resource_param='event_id', level=READ):
    """
    检查用户对资源的权限级别
    """
    def decorator(view_func):
        def wrapped_view(request, *args, **kwargs):
            # 1. 获取资源ID
            resource_id = kwargs.get(resource_param)
            if not resource_id:
                raise NotFoundError("资源ID未指定")
            
            # 2. 检查权限
            if not permission_service.check_permission(
                request.user, 
                resource_type, 
                resource_id, 
                level
            ):
                raise ForbiddenError("权限不足")
            
            return view_func(request, *args, **kwargs)
        return wrapped_view
    return decorator
```

```python
# core/monitoring.py
import time
from functools import wraps
from django.core.cache import cache
import logging

logger = logging.getLogger(__name__)

class PerformanceMonitor:
    """性能监控装饰器"""
    
    @staticmethod
    def monitor(threshold_ms=2000):
        """监控函数执行时间"""
        def decorator(func):
            @wraps(func)
            def wrapper(*args, **kwargs):
                start_time = time.time()
                
                try:
                    result = func(*args, **kwargs)
                    return result
                finally:
                    execution_time = (time.time() - start_time) * 1000
                    
                    # 记录性能指标
                    cache_key = f"perf:{func.__name__}"
                    samples = cache.get(cache_key, [])
                    samples.append(execution_time)
                    cache.set(cache_key, samples, 3600)
                    
                    # 超时告警
                    if execution_time > threshold_ms:
                        logger.warning(
                            f"{func.__name__}执行时间过长: {execution_time:.2f}ms"
                        )
                        # 发送告警（集成到您的告警系统）
                        alert_sentry(
                            message=f"{func.__name__} 慢查询",
                            level="warning",
                            extra={"execution_time": execution_time}
                        )
            
            return wrapper
        return decorator

# 使用示例
@PerformanceMonitor.monitor(threshold_ms=1000)
def expensive_query():
    """耗时查询"""
    # 复杂的数据库查询
    pass
```
